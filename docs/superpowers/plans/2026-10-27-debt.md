> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Fix the Usage review's deferred minors (U1–U9), v1's four known limitations (V1–V4), and the `.impeccable` leftovers. Live refresh comes next and relies on these fixes.

**Architecture:**
- Store fixes are local to `store/usage.go` and `store/sessions.go`.
- `store.SessionScan` is a new mutex-guarded, in-memory cache of session heads. It follows `UsageScan`'s pattern: `App` owns one, and the Sessions and Cleanup loads pass it to `Projects`.
- The view fixes stay in `view/usage.go`, `view/format.go` and `view/config.go`.

**Tech Stack:** Go 1.27 and the standard library (`sync`, `io`, `regexp`), MyGo `ui`, `ui.Tester`.

**Spec:** `docs/superpowers/specs/2026-10-27-debt-design.md`

## Global Constraints

- **Read-only:** the app never writes anything in `~/.claude`. Tests never read the real `~/.claude`; they use `fixture.New` or `t.TempDir()`.
- **Permission tests:** a test that relies on `chmod 0` skips when `os.Geteuid() == 0`, and restores the mode in `t.Cleanup`.
- **Exact UI strings:**
  - the toast `N transcripts could not be read`;
  - a partly priced cost `$X+`, with the tooltip `N calls unpriced`;
  - `—` for a row with only unpriced calls;
  - `No usage yet.`
- **Masking:**
  - A name is split on `-`, `_` and `.`, and at camelCase boundaries, then lowercased.
  - It is masked when a segment equals one of `key token secret secrets pass passwd password passphrase passcode auth authorization cookie sig signature credential credentials bearer`, or ends in `key token secret password cookie`, except `hotkey`.
  - The three name patterns are searched one at a time.
  - `secretKey` doesn't change.
- **Every commit:**
  - Before it, `gofmt -l .` prints nothing, `go vet ./...` exits 0, and `CGO_ENABLED=1 go test -race ./...` exits 0.
  - The message ends with `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`. No session URLs.

## Review Focus

No spec item says what happens in these five cases. Each one gets its own test:

1. **The Sessions load and the Cleanup load run `Projects` at once on one `SessionScan`.** There must be no data race. Test: `TestProjectsScanFromTwoLoads` (Task 3), run under `-race`.
2. **A header inside a flag, `--header=X-API-Key: v`, stays masked** under the new per-pattern scan. Test: the existing `TestScalarMasksSecretHeaders` (Task 4) must stay green.
3. **A Trash confirmation is pending when the usage scan lands with an unreadable file.** The confirmation stays. Test: `TestUsageUnreadableToastsOnce` (Task 2).
4. **A transcript grows between the stat and the read.** The bytes past the stat'ed size wait for the next scan, so they are not counted early, and no offset ever passes the stored size. Test: `TestUsageScanStopsAtStatSize` (Task 1).
5. **A merged project is keyed by the folder of its newest session.** The Sessions table's key stays stable. Test: `TestProjectsMergeSameCwd` (Task 3).

```go
// The bytes a transcript grows by between the stat and the read wait for
// the next scan, so the read offset never passes the size the scan keeps.
func TestUsageScanStopsAtStatSize(t *testing.T) {
	s, d, _ := newStore(t)
	base := costOf(must2(s.Usage(&UsageScan{}, nil)))
	recent := filepath.Join(d.Root, fixture.Other, fixture.Recent+".jsonl")
	late := fixture.UsageLine("/Users/me/other", "2026-10-26T10:00:00.000Z", "r9", "claude-haiku-4-5", `{"input_tokens":1000000}`) + "\n" // $1
	grown := false
	scan := UsageScan{open: func(p string) (*os.File, error) {
		if p == recent && !grown {
			grown = true
			f, err := os.OpenFile(p, os.O_APPEND|os.O_WRONLY, 0)
			if err != nil {
				return nil, err
			}
			f.WriteString(late)
			f.Close()
		}
		return os.Open(p)
	}}
	if got := costOf(must2(s.Usage(&scan, nil))); !near(got, base) {
		t.Errorf("first scan costs %v, want %v: it read past the size it stat'ed", got, base)
	}
	if got := costOf(must2(s.Usage(&scan, nil))); !near(got, base+1) {
		t.Errorf("second scan costs %v, want %v", got, base+1)

		t.Fatal(err)
	}

		t.Errorf("after the cut, cost %v, want %v", got, base)
	}
}
```

```go
// costText is a row's cost: "—" when none of its calls has a price, and a
// "+" after the amount when only some have.
func costText(u store.Usage) string {
	switch {
	case u.Cost == 0 && u.Unpriced > 0:
		return "—"
	case u.Unpriced > 0:
		return money(u.Cost) + "+"
	}
	return money(u.Cost)
}

}
```

```go
// tokenCount is a count of tokens, short: 950, 12.3K, 820.4M, 2.1B. The unit
// is picked after rounding, so 999,950 is 1.0M, not 1000.0K.
func tokenCount(n int64) string {
	if n < 1000 {
		return strconv.FormatInt(n, 10)
	}
	for _, u := range []struct {
		div float64
		sym string
	}{{1e3, "K"}, {1e6, "M"}} {
		if math.Round(float64(n)/u.div*10) < 10000 {
			return fmt.Sprintf("%.1f%s", float64(n)/u.div, u.sym)
		}
	}
	return fmt.Sprintf("%.1fB", float64(n)/1e9)
}
```

```go
// SessionScan keeps what earlier scans read from the head of each session,
// so a Rescan rereads only the sessions whose .jsonl changed. Its zero value
// is ready, and the Sessions and Cleanup loads may use it at once.
type SessionScan struct {
	mu    sync.Mutex
	heads map[string]sessionHead // by .jsonl path
	reads int                    // heads read from disk: tests count them
}

// sessionHead is what readHead found in a .jsonl of this size and mtime.
type sessionHead struct {
	size  int64
	mtime time.Time
	Session
}

// head fills x's head metadata from the cache, or from the file when it
// changed since it was read.
func (sc *SessionScan) head(file string, info fs.FileInfo, x *Session) {
	sc.mu.Lock()
	h, ok := sc.heads[file]
	sc.mu.Unlock()
	if !ok || h.size != info.Size() || !h.mtime.Equal(info.ModTime()) {
		h = sessionHead{size: info.Size(), mtime: info.ModTime()}
		if err := readHead(file, &h.Session); err != nil {
			h.Err = err.Error()
		}
		sc.mu.Lock()
		if sc.heads == nil {
			sc.heads = map[string]sessionHead{}
		}
		sc.heads[file] = h
		sc.reads++
		sc.mu.Unlock()
	}
	x.Cwd, x.Branch, x.Version = h.Cwd, h.Branch, h.Version
	x.Title, x.FirstPrompt, x.Started, x.Err = h.Title, h.FirstPrompt, h.Started, h.Err
}

// keep drops the heads of the files not in seen: sessions gone.
func (sc *SessionScan) keep(seen map[string]bool) {
	sc.mu.Lock()
	defer sc.mu.Unlock()
	for f := range sc.heads {
		if !seen[f] {
			delete(sc.heads, f)
		}
	}
}
```

```go
// secretValue matches credentials inside a string value by their shape,
// whatever its key.
// shortcut: known shapes only (bearer/basic, sk-, gh?_, github_pat_, xox?-, AKIA, user:pass@); a token without a known
// shape still shows unless secretString finds it named like a secret. Match more shapes if one leaks.
var secretValue = regexp.MustCompile(`(?i)bearer\s+\S+|basic\s+\S+|\bsk-[\w-]{8,}|\bgh[pousr]_\w+|github_pat_\w+|xox[abposr]-\S+|\bAKIA[0-9A-Z]{16}|://[^/\s:@]+:[^@\s]+@`)

// secretNamed find the names a string gives its values: a query parameter,
// a --flag=, a "Name: value" header. Each is searched on its own, as one can
// sit inside another (--header=X-API-Key: v).
var secretNamed = []*regexp.Regexp{
	regexp.MustCompile(`[?&]([^=&\s]+)=[^&\s]+`),
	regexp.MustCompile(`--?([\w-]+)=\S+`),
	regexp.MustCompile(`(?:^|[=\s])([\w-]+)\s*:\s*\S`),
}

// secretString reports whether s holds a credential: one of a known shape,
// or a value named like a secret. A bare value after a separate flag
// element outside an MCP "args" array still shows (inside one it is masked,
// see tree.add).
func secretString(s string) bool {
	if secretValue.MatchString(s) {
		return true
	}
	for _, re := range secretNamed {
		for _, m := range re.FindAllStringSubmatch(s, -1) {
			if secretName(m[1]) {
				return true
			}
		}
	}
	return false
}

// camelBoundary splits apiKey into api_Key.
var camelBoundary = regexp.MustCompile(`([a-z0-9])([A-Z])`)

// secretSegments are name segments that name a credential by themselves.
var secretSegments = map[string]bool{
	"key": true, "token": true, "secret": true, "secrets": true, "pass": true, "passwd": true, "password": true,
	"passphrase": true, "passcode": true, "auth": true, "authorization": true, "cookie": true, "sig": true,
	"signature": true, "credential": true, "credentials": true, "bearer": true,
}

// secretName reports whether a parameter, flag or header name names a
// credential. It looks at the name's segments, split on -, _, . and at
// camelCase: one that is a credential word, or that ends in key, token,
// secret, password or cookie (hotkey aside). So author, hotkey and design
// don't count, while x-api-key, privatekey and access_token do.
func secretName(name string) bool {
	name = strings.ToLower(camelBoundary.ReplaceAllString(name, "${1}_${2}"))
	for _, seg := range strings.FieldsFunc(name, func(r rune) bool { return r == '-' || r == '_' || r == '.' }) {
		if secretSegments[seg] {
			return true
		}
		if seg == "hotkey" {
			continue
		}
		for _, end := range []string{"key", "token", "secret", "password", "cookie"} {
			if strings.HasSuffix(seg, end) {
				return true
			}
		}
	}
	return false
}
```
